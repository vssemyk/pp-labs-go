# Лабораторная №2 - Работа с числами, циклами и структурами в Go

## Выполненные задания

1.	Написана программа, которая определяет, является ли введенное пользователем число четным или нечетным.
2.	Реализована функция, которая принимает число и возвращает "Positive", "Negative" или "Zero".
3.	Написана программу, которая выводит все числа от 1 до 10 с помощью цикла for.
4.	Написана функция, которая принимает строку и возвращает ее длину.
5.	Создана структура Rectangle и реализован метод для вычисления площади прямоугольника.
6.	Написана функция, которая принимает два целых числа и возвращает их среднее значение.

## Требования

- Go 1.20+

## Установка и запуск

Сохраните файл под именем lb2.go, перейдите в каталог с ним и выполните команду:

```bash
  go run lb2.go
```
После запуска выберите опцию из списка (1-6) для вызова одноименной функции. Для выхода из программы выберите опцию "0".
При выборе некорректного пункта выводится сообщение "Некорректный пункт" и заново отображается список опций.

## Задание 1 - Четность числа
Принимает число, вычисляет остаток от его деления на 2 и делает вывод о четности или нечетности.

### Пример 
`Введите число: 17`

### Ответ
`17  - нечетное число`

## Задание 2 - Определение положительности числа
Получает число, передает его в функцию isPositive и возвращает "Positive", "Zero" или "Negative" в зависимости от отклонения от нуля.

### Пример 
`Введите число: -78`

### Ответ
`Negative`

## Задание 4 - Длина строки

### Пример

## Задание 5 - Структура Rectangle
Создана структура Rectangle. При вызове опции "Площадь прямоугольника" получает от пользователя длину и ширину, передает их в структуру, откуда значения берет функция rectangleSquare, вычисляющая площадь при помощи умножения значений сторон.

### Структура Rectangle

```
{
  width, length float64
}
```

### Пример 
```
Введите длину: 5
Введите ширину: 12
```

### Ответ
```
Ширина =  12 , длина =  5
Площадь прямоугольника:  60
```

## Задание 6 - Среднее значение двух чисел
Реализована функция average, вычисляющая среднее арифметическое двух чисел типа float64.

### Пример 
```
Введите первое число: 6
Введите второе число: 8
```

### Ответ
`Среднее: 7`

## Исходный код

```go
package main

import (
	"bufio"
	"fmt"
	"os"
	"strings"
)

type rect struct {
	width, length float64
}

func isEven(num int) {
	if num%2 == 0 {
		fmt.Println(num, " - четное число")
	} else {
		fmt.Println(num, " - нечетное число")
	}

}

func isPositive(num int) string {
	if num > 0 {
		return "Positive"
	} else if num == 0 {
		return "Zero"
	} else {
		return "Negative"
	}
}

func stringLength(str string) int {
	return len([]rune(str))
}

func (rec rect) rectangleSquare() float64 {
	return rec.width * rec.length
}

func average(a, b int) float64 {
	return float64(a+b) / 2
}

func main() {

	reader := bufio.NewReader(os.Stdin)

	for {

		var num int = 0
		var option int
		fmt.Println("\n=====Выберите задание:===== \n 1. Четность\n 2. Положительность\n 3. Цикл for\n 4. Длина строки\n 5. Площадь прямоугольника\n 6. Среднее из двух\n 0. Закрыть программу")
		fmt.Scanf("%d\n", &option)

		switch option {
		case 1:

			fmt.Println("*****Определение четности числа*****")
			fmt.Print("Введите число: ")
			fmt.Scanf("%d\n", &num)

			isEven(num)

		case 2:
			fmt.Println("*****Определение положительности числа*****")
			fmt.Print("Введите число: ")
			fmt.Scanf("%d\n", &num)

			fmt.Println(isPositive(num))

		case 3:
			fmt.Println("*****Цикл for*****")

			for i := 1; i <= 10; i++ {
				fmt.Println(i)
			}

		case 4:

			fmt.Println("*****Определение длины строки*****")
			_, _ = reader.ReadString('\n')
			fmt.Print("Введите строку: ")
			text, _ := reader.ReadString('\n')
			text = strings.TrimRight(text, "\r\n")

			fmt.Println("Длина строки:", stringLength(text))

		case 5:

			var rec rect

			fmt.Println("*****Площадь прямоугольника*****")
			fmt.Print("Введите длину: ")
			fmt.Scanf("%f\n", &rec.length)
			fmt.Print("Введите ширину: ")
			fmt.Scanf("%f\n", &rec.width)

			fmt.Println("Ширина = ", rec.width, ", длина = ", rec.length)

			fmt.Println("Площадь прямоугольника: ", rec.rectangleSquare())

		case 6:
			var num1, num2 int
			fmt.Println("*****Среднее значение двух чисел*****")
			fmt.Print("Введите первое число: ")
			fmt.Scanf("%d\n", &num1)
			fmt.Print("Введите второе число: ")
			fmt.Scanf("%d\n", &num2)

			fmt.Println("Среднее: ", average(num1, num2))

		case 0:
			fmt.Println("Выход")
			return

		default:
			fmt.Println("\nНекорректный пункт")
		}
	}

}
```
